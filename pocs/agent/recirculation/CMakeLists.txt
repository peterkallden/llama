add_executable(llama-recirculation-poc main.cpp)
target_include_directories(llama-recirculation-poc PRIVATE ${PROJECT_SOURCE_DIR}/src ${PROJECT_SOURCE_DIR}/vendor)
target_link_libraries(llama-recirculation-poc PRIVATE llama)
target_compile_features(llama-recirculation-poc PRIVATE cxx_std_17)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${PROJECT_SOURCE_DIR}
    OUTPUT_VARIABLE RECIRCULATION_SOURCE_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
target_compile_definitions(llama-recirculation-poc PRIVATE RECIRCULATION_SOURCE_REVISION="${RECIRCULATION_SOURCE_REVISION}")
if (BUILD_TESTING)
    add_executable(test-agent-recirculation-contract ../test-recirculation-contract.cpp)
    add_executable(test-agent-recirculation-runner test-runner.cpp)
    add_executable(test-agent-recirculation-memory test-memory.cpp)
    target_include_directories(test-agent-recirculation-memory PRIVATE ${PROJECT_SOURCE_DIR}/src)
    target_link_libraries(test-agent-recirculation-memory PRIVATE llama)
    foreach(part contract runner memory)
        set(target test-agent-recirculation-${part})
        target_compile_features(${target} PRIVATE cxx_std_17)
        if (MSVC)
            target_compile_options(${target} PRIVATE /UNDEBUG)
        else()
            target_compile_options(${target} PRIVATE -UNDEBUG)
        endif()
        add_test(NAME llama-agent-recirculation-${part}-ctest COMMAND ${target})
        set_tests_properties(llama-agent-recirculation-${part}-ctest PROPERTIES LABELS "agent;recirculation")
    endforeach()
endif()
