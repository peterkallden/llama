set(TARGET llama-plan)
add_library(${TARGET} plan-context.cpp plan-context.h plan-in-memory.cpp plan-in-memory.h plan-json.cpp plan-json.h plan-policy.cpp plan-policy.h plan-store.h plan-template.cpp plan-template.h plan-types.h)
target_include_directories(${TARGET} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_compile_features(${TARGET} PUBLIC cxx_std_17)
target_link_libraries(${TARGET} PUBLIC llama-memory)
if (BUILD_SHARED_LIBS)
    set_target_properties(${TARGET} PROPERTIES POSITION_INDEPENDENT_CODE ON)
    set_target_properties(${TARGET} PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()
if (LLAMA_PLAN_COZO)
    if (NOT COZO_INCLUDE_DIR)
        find_path(COZO_INCLUDE_DIR cozo_c.h)
    endif()
    if (NOT COZO_LIBRARY)
        find_library(COZO_LIBRARY NAMES cozo_c cozo)
    endif()
    if (NOT COZO_INCLUDE_DIR OR NOT COZO_LIBRARY)
        message(FATAL_ERROR "LLAMA_PLAN_COZO=ON requires Cozo C API headers and library. Set COZO_INCLUDE_DIR and COZO_LIBRARY.")
    endif()
    target_sources(${TARGET} PRIVATE cozo/plan-cozo.cpp cozo/plan-cozo.h cozo/plan-cozo-schema.cpp cozo/plan-cozo-schema.h)
    target_compile_definitions(${TARGET} PUBLIC LLAMA_PLAN_USE_COZO)
    target_include_directories(${TARGET} PRIVATE ${COZO_INCLUDE_DIR})
    target_link_libraries(${TARGET} PRIVATE ${COZO_LIBRARY})
    if (WIN32)
        target_link_libraries(${TARGET} PRIVATE ws2_32 bcrypt ntdll rpcrt4 uuid shlwapi crypt32)
    endif()
endif()
